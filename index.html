        SQL cheatsheet
        
-- Move east
select game.move(1, 0);

-- Query carried items
select item.id, item.kind, kind.name
from item.items item join item.kind kind using (kind)
where item.holder_id = (select id from creature.creatures where kind = 'hero');

-- Find visible items on the floor
select item.id, item.kind, kind.name, item.x, item.y
from item.items item join item.kind kind using (kind)
where item.holder_id is null and creature.in_sight(item.x, item.y);

-- Find floor items next to you that you can pick up
select item.id, item.kind, kind.name
from item.items item
join item.kind kind using (kind)
cross join creature.hero() hero
where item.holder_id is null
  and abs(item.x - hero.x) + abs(item.y - hero.y) <= 1;

-- Pick up an adjacent item (replace 17 with its id)
select game.pick_up(17);

-- Attack a visible rat with your sword
select game.attack(
  (select id from item.items
   where kind = 'sword'
     and holder_id = (select id from creature.creatures where kind = 'hero')
   limit 1),
  (select target.id from creature.creatures target
   where target.kind = 'rat' and creature.in_sight(target.x, target.y)
   limit 1)
);
